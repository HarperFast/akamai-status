### TEST 1: Default Configuration (No Config)

**Configuration:** Uses defaults,
**Formula:** `utilization × 1000`  
**Load Test:** 200 concurrent requests, 10ms delay

### Load Script

```javascript
const https = require('https');

const TARGET = 'pilot04-us-lax01.harperdbcloud.com';
const CONCURRENT = 200;
const DELAY = 10;
const AUTH = Buffer.from('HDB_ADMIN:***').toString('base64');

const agent = new https.Agent({ keepAlive: true, rejectUnauthorized: false });

let total = 0, success = 0, errors = 0;
const start = Date.now();

setInterval(() => {
  const elapsed = ((Date.now() - start) / 1000).toFixed(1);
  const rps = Math.round(total / elapsed);
  console.log(`[${new Date().toISOString()}] ${elapsed}s | Requests: ${total} | RPS: ~${rps} | Success: ${success} | Errors: ${errors}`);
}, 5000);

setInterval(async () => {
  try {
    const utilRaw = await new Promise((resolve, reject) => {
      const req = https.request({
        hostname: TARGET,
        port: 9925,
        path: '/',
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Basic ${AUTH}`
        },
        rejectUnauthorized: false
      }, res => {
        let data = '';
        res.on('data', chunk => data += chunk);
        res.on('end', () => resolve(data));
      });
      req.on('error', reject);
      req.write(JSON.stringify({
        operation: 'search_by_conditions',
        database: 'system',
        table: 'hdb_analytics',
        conditions: [{ search_attribute: 'metric', search_value: 'utilization' }],
        sort: { attribute: 'id', descending: true },
        limit: 1,
        get_attributes: ['utilization']
      }));
      req.end();
    });

    const loadRaw = await new Promise((resolve, reject) => {
      https.get(`https://${TARGET}/current_load_for_gtm`, { rejectUnauthorized: false }, res => {
        let data = '';
        res.on('data', chunk => data += chunk);
        res.on('end', () => resolve(data));
      }).on('error', reject);
    });

    const utilData = JSON.parse(utilRaw);
    const util = utilData[0]?.utilization || 0;
    const expected = Math.round(util * 1000);
    
    console.log(`>>> MONITOR [${new Date().toISOString()}]`);
    console.log(`    Utilization (raw): ${utilRaw.trim()}`);
    console.log(`    Load (raw): ${loadRaw}`);
    console.log(`    Expected: ${expected} | Match: ${loadRaw.includes(expected.toString()) ? '✓' : '✗'}`);
  } catch (e) {
    console.log(`>>> MONITOR ERROR: ${e.message}`);
  }
}, 10000);

async function hammer() {
  while (true) {
    const batch = Array(CONCURRENT).fill().map(() => 
      new Promise(resolve => {
        https.get(`https://${TARGET}/Hello`, { agent }, res => {
          res.on('data', () => {});
          res.on('end', () => { total++; success++; resolve(); });
        }).on('error', () => { total++; errors++; resolve(); });
      })
    );
    await Promise.all(batch);
    await new Promise(r => setTimeout(r, DELAY));
  }
}

console.log(`[${new Date().toISOString()}] Starting: ${CONCURRENT} concurrent, ${DELAY}ms delay`);
console.log(`[${new Date().toISOString()}] Monitoring utilization + load every 10s\n`);
hammer();
```
